# Implementation Plan: MCP SDK Migration Test Suite

## Context

Feast currently uses `fastapi_mcp` to auto-wrap its FastAPI endpoints as MCP tools. We're migrating to a purpose-built MCP SDK server. This test suite:
- Covers every endpoint on both the feature server (11 endpoints) and registry server (59 endpoints)
- Compares MCP SDK tool output vs direct REST output (exact match for reads, side-effect verification for writes)
- Tracks migration progress with an automated checklist printed after every test run
- Validates authorization (401/403) with proper error codes

### Step 1.2: Write `manifest.py` — core dataclasses

Uses `enum.Enum` for `ServerType` and `HttpMethod` instead of raw strings. All fields are explicitly typed.

```python
from dataclasses import dataclass, field
from enum import Enum

class ServerType(Enum):
    FEATURE_SERVER = "feature_server"
    REGISTRY = "registry"

class HttpMethod(Enum):
    GET = "GET"
    POST = "POST"
    DELETE = "DELETE"

@dataclass(frozen=True)
class ToolTestSpec:
    tool_name: str
    feast_endpoint: str
    http_method: HttpMethod
    server_type: ServerType
    sample_inputs: list[dict[str, Any]]
    migrated: bool = False
    mutation: bool = False
    verify_endpoint: str | None = None
    verify_expected: dict[str, Any] | None = None

@dataclass(frozen=True)
class DiscoveredEndpoint:
    path: str
    method: str
    server_type: ServerType
    request_schema: dict[str, Any]
    response_schema: dict[str, Any]
    parameters: list[dict[str, Any]]
```

Plus two lists: `FEATURE_SERVER_TOOLS` and `REGISTRY_TOOLS`.

### Step 1.3: Write the test feature repo

**`feature_store.yaml`** — SQLite online store, file-based offline store, file registry, MCP enabled:
- Modeled after `examples/mcp_feature_store/feature_store.yaml`
- `project: mcp_test_project`
- `provider: local`
- `online_store.type: sqlite`
- `offline_store.type: file`
- `feature_server.type: mcp`, `mcp_enabled: true`, `mcp_transport: http`
- No auth initially (auth tests use a separate config)

**`definitions.py`** — Feature definitions using existing Feast patterns:
- Entity: `driver` (join_key: `driver_id`, value_type: INT64)
- Data source: `FileSource` pointing to `data/driver_stats.parquet`
- Feature view: `driver_hourly_stats` with features: `conv_rate` (FLOAT), `acc_rate` (FLOAT), `avg_daily_trips` (INT64)
- Feature service: `driver_activity` referencing `driver_hourly_stats`
- Push source: `driver_stats_push` for push/write mutation tests
- Generate `data/driver_stats.parquet` with ~20 rows for driver_ids 1001-1020

**`permissions.py`** — RBAC definitions for auth tests:
- `admin_permission`: roles=["admin"], all actions on FeatureView + Entity
- `reader_permission`: roles=["reader"], DESCRIBE + READ_ONLINE only
- Reuse pattern from `examples/mcp_auth_feature_store/feature_repo/permissions.py`

def test_feature_server_coverage(feature_server: str) -> None:
    discovered: list[DiscoveredEndpoint] = auto_discover_endpoints(
        f"{feature_server}/openapi.json", ServerType.FEATURE_SERVER
    )
    manifest_set: set[tuple[str, str]] = {
        (t.feast_endpoint, t.http_method.value) for t in FEATURE_SERVER_TOOLS
    }
    discovered_set: set[tuple[str, str]] = {
        (d.path, d.method) for d in discovered
    }
    uncovered: set[tuple[str, str]] = discovered_set - manifest_set
    stale: set[tuple[str, str]] = manifest_set - discovered_set
    if uncovered:
        warnings.warn(f"Uncovered endpoints: {uncovered}")

### Step 4.2: `test_feature_server.py` — Parametrized tests

```python
@pytest.mark.parametrize("spec", FEATURE_SERVER_TOOLS, ids=lambda s: s.tool_name)
def test_feature_server_tool(spec: ToolTestSpec, feature_server: str) -> None:
    for sample_input in spec.sample_inputs:
        rest_result: httpx.Response = call_rest(
            feature_server, spec.feast_endpoint, spec.http_method, sample_input
        )
        assert rest_result.status_code in (200, 201, 202)

        if not spec.migrated:
            continue

        mcp_result: McpToolResult = call_mcp_tool(
            f"{feature_server}/mcp", spec.tool_name, sample_input
        )

        if spec.mutation:
            assert not mcp_result.is_error
            if spec.verify_endpoint:
                verify: httpx.Response = call_rest(
                    feature_server, spec.verify_endpoint, HttpMethod.GET
                )
                # assert expected data present
        else:
            assert normalize_response(rest_result.json()) == normalize_response(
                mcp_result.content
            )
```

### Step 4.4: `test_authorization.py` — Auth cases

```python
def test_no_token_returns_401(auth_feature_server: str) -> None:
    resp: httpx.Response = call_rest(
        auth_feature_server, "/get-online-features", HttpMethod.POST,
        payload=SAMPLE_GET_ONLINE_PAYLOAD, headers={},
    )
    assert resp.status_code == 401

def test_invalid_token_returns_401(auth_feature_server: str) -> None:
    resp: httpx.Response = call_rest(
        ..., headers={"Authorization": "Bearer invalid.jwt.token"},
    )
    assert resp.status_code == 401

def test_insufficient_role_returns_403(
    auth_feature_server: str, mock_oidc_server: str
) -> None:
    token: str = get_token(mock_oidc_server, "user")
    resp: httpx.Response = call_rest(
        ..., headers={"Authorization": f"Bearer {token}"},
    )
    assert resp.status_code == 403

def test_valid_role_returns_200(
    auth_feature_server: str, mock_oidc_server: str
) -> None:
    token: str = get_token(mock_oidc_server, "admin")
    resp: httpx.Response = call_rest(
        ..., headers={"Authorization": f"Bearer {token}"},
    )
    assert resp.status_code == 200
```

Auth tests run against both feature server and registry server endpoints.
